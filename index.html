        ${STARTUPS.map(s=>{const a=row.top_alloc?.[s.id]||0;return a>0?`<span class="tag"><b>${s.name}</b>: ${fmt(a)}</span>`:''}).join('')}

      </div>

      <div style="margin-bottom:.75rem">

        <label style="margin-bottom:4px">Expense domain profile (where their mind goes)</label>

        <div style="display:flex;gap:6px;flex-wrap:wrap">

          ${DOMAINS.map(d=>{

            const amt=db2[d]||0; const pct2=totalExp>0?Math.round(amt/totalExp*100):0;

            return pct2>0?`<div style="background:#EEF2FF;border-radius:6px;padding:4px 8px;font-size:11px"><b style="color:#4F46E5">${d}</b><br><span style="color:#555">${pct2}% (${fmt(amt)})</span></div>`:'';

          }).join('')}

        </div>

      </div>

      ${row.crisis_call?`<div style="margin-bottom:.5rem">${Object.entries(row.crisis_call).map(([sid,v])=>{const st=STARTUPS.find(x=>x.id===sid);return`<span class="tag"><b>${st?.name||sid}:</b> ${v.call}</span>`;}).join('')}</div>`:''}

      <details>

        <summary style="font-size:12px;cursor:pointer;color:#4F46E5">Full detail view</summary>

        <div style="margin-top:.75rem;font-size:12px;line-height:1.7">

          <b>Diagnostic answers:</b><br>

          ${STARTUPS.map(s=>{

            const d2=row.diagnostics?.[s.id]||{};

            const qs=DIAGNOSTICS[s.id];

            if(!Object.keys(d2).length)return'';

            return`<div style="margin:.4rem 0"><b>${s.name}:</b> ${qs.map((q,i)=>`[${q.q.substring(0,30)}...] → ${d2['q'+i]||'—'}`).join(' | ')}</div>`;

          }).join('')}

          <br><b>Expenses breakdown:</b><br>

          ${STARTUPS.map(s=>{

            const ex=row.expenses?.[s.id]||{};

            const items=Object.entries(ex).filter(([,v])=>v>0);

            if(!items.length)return'';

            return`<div style="margin:.4rem 0"><b>${s.name}:</b> ${items.map(([eid,amt])=>{const exp=EXPENSES.find(e=>e.id===eid);return`${exp?.label||eid} (${fmt(amt)})`;}).join(' · ')}</div>`;

          }).join('')}

          <br><b>Crisis reasoning:</b><br>

          ${Object.entries(row.crisis_call||{}).map(([sid,v])=>{const st=STARTUPS.find(x=>x.id===sid);return`<div style="margin:.4rem 0"><b>${st?.name||sid} — ${v.call}:</b> ${v.reason}</div>`;}).join('')}

        </div>

      </details>

    </div>`;}).join('')}`;

}
